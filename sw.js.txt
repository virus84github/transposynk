// Service Worker per TranspoSync PWA (Real-Time Cloud & Network-First)
const CACHE_NAME = 'transposync-live-v3';

// Installazione: forza l'attivazione immediata senza attendere la chiusura delle schede
self.addEventListener('install', (event) => {
  self.skipWaiting();
});

// Attivazione: distrugge immediatamente qualsiasi vecchia cache rimasta nei browser di tutti i dispositivi
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) => {
      return Promise.all(
        keys.map((key) => {
          if (key !== CACHE_NAME) {
            return caches.delete(key);
          }
        })
      );
    }).then(() => self.clients.claim())
  );
});

// Fetch: strategia Network-First per HTML e dati live, garantendo che tutti i colleghi ricevano sempre i dati aggiornati
self.addEventListener('fetch', (event) => {
  // Per navigazione e documenti HTML: cerca sempre prima la versione remota più recente
  if (event.request.mode === 'navigate' || event.request.destination === 'document') {
    event.respondWith(
      fetch(event.request)
        .catch(() => caches.match(event.request) || caches.match('./index.html'))
    );
    return;
  }

  // Per risorse statiche e CDN
  event.respondWith(
    caches.match(event.request).then((cachedResponse) => {
      if (cachedResponse) {
        return cachedResponse;
      }
      return fetch(event.request).then((networkResponse) => {
        if (networkResponse && networkResponse.status === 200 && event.request.method === 'GET') {
          const responseToCache = networkResponse.clone();
          caches.open(CACHE_NAME).then((cache) => {
            cache.put(event.request, responseToCache);
          });
        }
        return networkResponse;
      }).catch(() => {
        // Fallback offline silenzioso
      });
    })
  );
});
